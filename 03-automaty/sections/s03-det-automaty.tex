\section{Deterministické konečné automaty}\label{sec:det-automaty}

\begin{definition}[Deterministický konečný automat]
    \emph{Deterministický konečný automat} (zkráceně DFA) je uspořádaná pětice
    \[A=(Q,\Sigma,\delta,q_0,F),\]
    kde
    \begin{itemize}
        \item $Q$ je konečná množina stavů,
        \item $\Sigma$ je vstupní abeceda,
        \item $\delta:Q\times\Sigma\to Q$ je přechodová funkce,~která každé dvojici stavu a~znaku přiřazuje právě jeden následující stav,
        \item $q_0\in Q$ je počáteční stav a
        \item $F\subseteq Q$ je množina přijímajících stavů.
    \end{itemize}
\end{definition}

Při kreslení někdy kvůli přehlednosti vynecháme \emph{odpadní stav}: všechny nezakreslené přechody si pak představujeme vedené do tohoto nepřijímajícího stavu,~který má smyčku pro každý znak. Formálně je tak přechodová funkce stále definována pro každou dvojici.

Automat zakreslujeme jako orientovaný graf. Stavy jsou vrcholy,~přechody hrany s~popisky. Počáteční stav označíme šipkou vedoucí \uv{odnikud} a~přijímající stavy dvojitým okrajem.

\subsection{Výpočet automatu}

Na začátku se automat nachází ve stavu $q_0$. Vstupní slovo čte zleva doprava. Nachází-li se ve stavu $q$ a~čte znak $a$,~přejde do stavu $\delta(q,a)$. Po přečtení celého slova přijme právě tehdy,~když skončil ve stavu z~$F$.

\begin{definition}[Jazyk automatu]
    Jazykem automatu $A$ nad abecedou $\Sigma$ rozumíme množinu
    \[L(A)=\set{w\in\Sigma^*:\text{$A$ přijme slovo $w$}}.\]
\end{definition}

\begin{example}[Slova začínající na $01$]\label{ex:dfa-prefix}
    Automat na obrázku \ref{fig:dfa-prefix} nejprve vyžaduje znaky $0$ a~$1$. Jakmile je přečte,~zůstane v~přijímajícím stavu bez ohledu na zbytek vstupu.
    \begin{figure}[h]
        \centering
        \input{03-automaty/images/ch03_dfa-prefix-01.tex}
        \caption{DFA přijímající právě slova začínající na $01$.}
        \label{fig:dfa-prefix}
    \end{figure}
    Jeho jazyk je
    \[L(A)=\set{01u:u\in\set{0,1}^*}.\]
    Slovo $0110$ přijme,~zatímco $001$ odmítne už při čtení druhého znaku.
\end{example}

\subsection{Složitější příklady}

\begin{example}[Slova končící na $01$]\label{ex:dfa-suffix}
    Nyní už nestačí zkontrolovat první dva znaky. Automat si musí průběžně pamatovat nejdelší konec dosud přečtené části,~který by mohl být začátkem slova $01$:
    \begin{itemize}
        \item $q_0$: poslední znak není $0$ nebo jsme zatím nic nepřečetli;
        \item $q_1$: poslední znak je $0$;
        \item $q_2$: poslední dva znaky jsou $01$.
    \end{itemize}
    \begin{figure}[h]
        \centering
        \input{03-automaty/images/ch03_dfa-suffix-01.tex}
        \caption{DFA pro jazyk $L_{\mathrm{suff}}=\set{u01:u\in\set{0,1}^*}$.}
        \label{fig:dfa-suffix}
    \end{figure}
\end{example}

\begin{example}[Sudý počet nul a~lichý počet jedniček]\label{ex:dfa-parita}
    Potřebujeme si pamatovat dvě informace,~každou se dvěma možnostmi. Dostaneme proto čtyři stavy $q_{SS},q_{SL},q_{LS},q_{LL}$; první index udává paritu počtu nul a~druhý paritu počtu jedniček. Znak $0$ změní první index,~znak $1$ druhý.
    \begin{figure}[h]
        \centering
        \input{03-automaty/images/ch03_dfa-parita.tex}
        \caption{DFA přijímající slova se sudým počtem nul a~lichým počtem jedniček.}
        \label{fig:dfa-parita}
    \end{figure}
    Jediným přijímajícím stavem je $q_{SL}$.
\end{example}

\begin{example}[Binární čísla dělitelná třemi]\label{ex:dfa-delitelnost}
    Čteme-li binární číslo zleva doprava a~dosud přečtená část dává po dělení třemi zbytek $r$,~pak po připsání bitu $b\in\set{0,1}$ dostaneme zbytek čísla $2r+b$. Stačí si tedy pamatovat zbytek $0$,~$1$ nebo $2$.
    \begin{figure}[h]
        \centering
        \input{03-automaty/images/ch03_dfa-delitelnost-3.tex}
        \caption{DFA pro neprázdné binární zápisy čísel dělitelných třemi.}
        \label{fig:dfa-delitelnost}
    \end{figure}
    Kvůli vyloučení prázdného slova používáme zvláštní počáteční stav $s$. Automat například přijme $110_2=6$,~ale odmítne $101_2=5$.
\end{example}
